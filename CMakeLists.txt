cmake_minimum_required(VERSION 3.16)
project(ThreeBodySimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(threebody_core STATIC
  core/physics/Gravity.cpp core/physics/RK4Integrator.cpp
  core/physics/Invariants.cpp core/physics/InitialConditions.cpp
  renderer/null/NullRenderBackend.cpp)
target_include_directories(threebody_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(threebody_core PRIVATE /O2)
else()
  target_compile_options(threebody_core PRIVATE -O3 -fno-fast-math)
endif()
add_library(threebody_simulator STATIC core/simulation/Simulator.cpp core/simulation/TrajectoryJournal.cpp)
target_link_libraries(threebody_simulator PUBLIC threebody_core Threads::Threads)
option(THREEBODY_BUILD_HOST_TESTS "Build simulator correctness tests" ON)
if(THREEBODY_BUILD_HOST_TESTS)
  enable_testing()
  foreach(name vsync_pacing frame_rate_journal frame_generation pipeline scene_mesh demo_validation simulator trajectory long_lived_triple stable_triple random_triple returning_triple rk45 planet frame_pacing)
    if(name STREQUAL "demo_validation")
      add_executable(${name}_test tests/demo_validation.cpp)
    else()
      add_executable(${name}_test tests/${name}_test.cpp)
    endif()
    target_link_libraries(${name}_test PRIVATE threebody_simulator)
    # Assertions must execute even in Release; several tests call APIs in assert.
    if(MSVC)
      target_compile_options(${name}_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name}_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
    set_tests_properties(${name} PROPERTIES TIMEOUT 90)
  endforeach()
endif()
if(THREEBODY_BUILD_HOST_TESTS)
  add_executable(planet_runtime_test tests/planet_runtime_test.cpp core/simulation/Simulator.cpp core/simulation/TrajectoryJournal.cpp)
  target_compile_definitions(planet_runtime_test PRIVATE THREEBODY_DETAILED=1 THREEBODY_PLANET=1)
  target_link_libraries(planet_runtime_test PRIVATE threebody_core Threads::Threads)
  add_test(NAME planet_runtime COMMAND planet_runtime_test)
  add_executable(detailed_trajectory_test tests/detailed_trajectory_test.cpp core/simulation/Simulator.cpp core/simulation/TrajectoryJournal.cpp)
  target_compile_definitions(detailed_trajectory_test PRIVATE THREEBODY_DETAILED=1)
  target_link_libraries(detailed_trajectory_test PRIVATE threebody_core Threads::Threads)
  add_test(NAME detailed_trajectory COMMAND detailed_trajectory_test)
  set_tests_properties(planet_runtime detailed_trajectory PROPERTIES TIMEOUT 90)
endif()
